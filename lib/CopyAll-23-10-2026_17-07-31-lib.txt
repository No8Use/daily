import 'dart:convert';
import 'dart:io';
import 'dart:typed_data';

import 'package:daily/data/database/app_database.dart';
import 'package:daily/utils/date_utils.dart';
import 'package:drift/drift.dart';
import 'package:file_picker/file_picker.dart';
import 'package:intl/intl.dart';

enum ImportMode { merge, replace }

class BackupService {
  BackupService(this.db);

  final AppDatabase db;

  /// Exports the database to JSON and opens the system save dialog.
  Future<void> exportData() async {
    final habits = await db.select(db.habits).get();
    final completions = await db.select(db.habitCompletions).get();
    final tasks = await db.select(db.tasks).get();
    final notes = await db.select(db.notes).get();
    final reviews = await db.select(db.dailyReviews).get();

    final exportJson = {
      'version': 1,
      'exportedAt': DateTime.now().toIso8601String(),
      'habits': habits
          .map((h) => {
        'id': h.id,
        'title': h.title,
        'description': h.description,
        'createdAt': h.createdAt.toIso8601String(),
        'canceledAt': h.canceledAt?.toIso8601String(),
      })
          .toList(),
      'habitCompletions': completions
          .map((c) => {
        'id': c.id,
        'habitId': c.habitId,
        'date': AppDateUtils.toIsoDateString(c.date),
        'isCompleted': c.isCompleted,
      })
          .toList(),
      'tasks': tasks
          .map((t) => {
        'id': t.id,
        'title': t.title,
        'description': t.description,
        'createdAt': t.createdAt.toIso8601String(),
        'dueDate': t.dueDate == null
            ? null
            : AppDateUtils.toIsoDateString(t.dueDate!),
        'isCompleted': t.isCompleted,
      })
          .toList(),
      'notes': notes
          .map((n) => {
        'id': n.id,
        'title': n.title,
        'description': n.description,
        'createdAt': n.createdAt.toIso8601String(),
        'needCheckDate': n.needCheckDate?.toIso8601String(),
      })
          .toList(),
      'dailyReviews': reviews
          .map((r) => {
        'id': r.id,
        'date': AppDateUtils.toIsoDateString(r.date),
        'description': r.description,
        'rating': r.rating,
      })
          .toList(),
    };

    final timestamp = DateFormat('yyyy-MM-dd_HH-mm').format(DateTime.now());
    final fileName = 'daily_backup_$timestamp.json';
    final jsonString = const JsonEncoder.withIndent('  ').convert(exportJson);

    await FilePicker.saveFile(
      dialogTitle: 'Сохранить резервную копию Daily',
      fileName: fileName,
      type: FileType.custom,
      allowedExtensions: ['json'],
      bytes: Uint8List.fromList(utf8.encode(jsonString)),
    );
  }

  /// Opens a system file picker and reads a JSON backup.
  Future<Map<String, dynamic>?> pickBackupFile() async {
    final result = await FilePicker.pickFiles(
      type: FileType.custom,
      allowedExtensions: ['json'],
      // withData: true,
    );

    if (result.length != 1) {
      return null;
    }

    final pickedFile = result.single;
    final bytes = await pickedFile.readAsBytes();

    final content = utf8.decode(bytes);
    final decoded = jsonDecode(content);

    if (decoded is! Map<String, dynamic>) {
      throw const FormatException(
        'Файл резервной копии должен содержать JSON-объект.',
      );
    }

    if (decoded['version'] != 1) {
      throw FormatException(
        'Неподдерживаемая версия резервной копии: ${decoded['version']}',
      );
    }

    return decoded;
  }

  /// Imports backup data using merge or replace mode.
  Future<void> importData(
      Map<String, dynamic> data,
      ImportMode mode,
      ) async {
    await db.transaction(() async {
      if (mode == ImportMode.replace) {
        // Delete dependent records before their parent records.
        await db.delete(db.habitCompletions).go();
        await db.delete(db.habits).go();
        await db.delete(db.tasks).go();
        await db.delete(db.notes).go();
        await db.delete(db.dailyReviews).go();
      }

      final habitsList = (data['habits'] as List?) ?? [];
      for (final item in habitsList) {
        final companion = HabitsCompanion(
          id: Value(item['id'] as int),
          title: Value(item['title'] as String),
          description: Value(item['description'] as String?),
          createdAt: Value(DateTime.parse(item['createdAt'] as String)),
          canceledAt: Value(
            item['canceledAt'] == null
                ? null
                : DateTime.parse(item['canceledAt'] as String),
          ),
        );

        await db.into(db.habits).insertOnConflictUpdate(companion);
      }

      final completionsList = (data['habitCompletions'] as List?) ?? [];
      for (final item in completionsList) {
        final companion = HabitCompletionsCompanion(
          id: Value(item['id'] as int),
          habitId: Value(item['habitId'] as int),
          date: Value(
            AppDateUtils.parseIsoDateString(item['date'] as String),
          ),
          isCompleted: Value(item['isCompleted'] as bool),
        );

        await db
            .into(db.habitCompletions)
            .insertOnConflictUpdate(companion);
      }

      final tasksList = (data['tasks'] as List?) ?? [];
      for (final item in tasksList) {
        final companion = TasksCompanion(
          id: Value(item['id'] as int),
          title: Value(item['title'] as String),
          description: Value(item['description'] as String?),
          createdAt: Value(DateTime.parse(item['createdAt'] as String)),
          dueDate: Value(
            item['dueDate'] == null
                ? null
                : AppDateUtils.parseIsoDateString(
              item['dueDate'] as String,
            ),
          ),
          isCompleted: Value(item['isCompleted'] as bool),
        );

        await db.into(db.tasks).insertOnConflictUpdate(companion);
      }

      final notesList = (data['notes'] as List?) ?? [];
      for (final item in notesList) {
        final companion = NotesCompanion(
          id: Value(item['id'] as int),
          title: Value(item['title'] as String),
          description: Value(item['description'] as String),
          createdAt: Value(DateTime.parse(item['createdAt'] as String)),
          needCheckDate: Value(
            item['needCheckDate'] == null
                ? null
                : DateTime.parse(item['needCheckDate'] as String),
          ),
        );

        await db.into(db.notes).insertOnConflictUpdate(companion);
      }

      final reviewsList = (data['dailyReviews'] as List?) ?? [];
      for (final item in reviewsList) {
        final companion = DailyReviewsCompanion(
          id: Value(item['id'] as int),
          date: Value(
            AppDateUtils.parseIsoDateString(item['date'] as String),
          ),
          description: Value(item['description'] as String),
          rating: Value(item['rating'] as int),
        );

        await db.into(db.dailyReviews).insertOnConflictUpdate(companion);
      }
    });
  }
}

import 'package:intl/intl.dart';

/// Date utility functions for date normalization and formatting.
class AppDateUtils {
  /// Normalizes a DateTime object to YYYY-MM-DD (start of day: 00:00:00).
  static DateTime normalizeDate(DateTime date) {
    return DateTime(date.year, date.month, date.day);
  }

  /// Formats a DateTime into ISO date string 'YYYY-MM-DD'.
  static String toIsoDateString(DateTime date) {
    final normalized = normalizeDate(date);
    return DateFormat('yyyy-MM-dd').format(normalized);
  }

  /// Parses 'YYYY-MM-DD' string back into normalized DateTime.
  static DateTime parseIsoDateString(String dateStr) {
    return normalizeDate(DateFormat('yyyy-MM-dd').parse(dateStr));
  }

  /// Formats date for UI display (e.g., '14 Oct 2026' or '14.10.2026').
  static String formatDateDisplay(DateTime date) {
    return DateFormat('dd.MM.yyyy').format(date);
  }

  /// Formats date with full month name for headers (e.g., '14 October 2026').
  static String formatDateFullHeader(DateTime date) {
    return DateFormat('d MMMM yyyy').format(date);
  }

  /// Checks if two dates represent the same calendar day.
  static bool isSameDay(DateTime? date1, DateTime? date2) {
    if (date1 == null || date2 == null) return false;
    return date1.year == date2.year &&
        date1.month == date2.month &&
        date1.day == date2.day;
  }

  /// Returns today's date normalized.
  static DateTime get today => normalizeDate(DateTime.now());
}

import 'dart:io';

import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:timezone/data/latest_all.dart' as tz;
import 'package:timezone/timezone.dart' as tz;

class NotificationService {
  NotificationService({
    FlutterLocalNotificationsPlugin? notificationsPlugin,
  }) : _notificationsPlugin =
      notificationsPlugin ?? FlutterLocalNotificationsPlugin();

  final FlutterLocalNotificationsPlugin _notificationsPlugin;

  static const String _channelId = 'daily_notes_channel';
  static const String _channelName = 'Напоминания о заметках';
  static const String _channelDescription =
      'Уведомления для запланированных заметок';

  /// Initializes local notifications and timezone data.
  Future<void> init() async {
    tz.initializeTimeZones();

    try {
      tz.setLocalLocation(tz.getLocation('Europe/Moscow'));
    } catch (_) {
      tz.setLocalLocation(tz.getLocation('UTC'));
    }

    const settings = InitializationSettings(
      android: AndroidInitializationSettings('@mipmap/ic_launcher'),
      iOS: DarwinInitializationSettings(
        requestAlertPermission: false,
        requestBadgePermission: false,
        requestSoundPermission: false,
      ),
    );

    await _notificationsPlugin.initialize(settings: settings);
  }

  /// Requests notification permissions on Android and iOS.
  Future<bool> requestPermissions() async {
    if (Platform.isAndroid) {
      final android = _notificationsPlugin
          .resolvePlatformSpecificImplementation<
          AndroidFlutterLocalNotificationsPlugin>();

      return await android?.requestNotificationsPermission() ?? false;
    }

    if (Platform.isIOS) {
      final ios = _notificationsPlugin
          .resolvePlatformSpecificImplementation<
          IOSFlutterLocalNotificationsPlugin>();

      return await ios?.requestPermissions(
        alert: true,
        badge: true,
        sound: true,
      ) ??
          false;
    }

    return true;
  }

  /// Schedules a notification for a note at [scheduledDate].
  Future<void> scheduleNoteNotification({
    required int noteId,
    required String title,
    required String body,
    required DateTime scheduledDate,
  }) async {
    if (scheduledDate.isBefore(DateTime.now())) {
      return;
    }

    final hasPermission = await requestPermissions();
    if (!hasPermission) {
      return;
    }

    const notificationDetails = NotificationDetails(
      android: AndroidNotificationDetails(
        _channelId,
        _channelName,
        channelDescription: _channelDescription,
        importance: Importance.high,
        priority: Priority.high,
      ),
      iOS: DarwinNotificationDetails(
        presentAlert: true,
        presentBadge: true,
        presentSound: true,
      ),
    );

    final scheduledTime = tz.TZDateTime.from(
      scheduledDate,
      tz.local,
    );

    await _notificationsPlugin.zonedSchedule(
      id: noteId,
      title: title,
      body: body,
      scheduledDate: scheduledTime,
      notificationDetails: notificationDetails,
      androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
    );
  }

  /// Cancels a scheduled notification by note ID.
  Future<void> cancelNoteNotification(int noteId) async {
    await _notificationsPlugin.cancel(id: noteId);
  }

  /// Cancels all scheduled notifications.
  Future<void> cancelAllNotifications() async {
    await _notificationsPlugin.cancelAll();
  }
}

/// Initializes GetIt service locator dependencies.
Future<void> setupServiceLocator() async {
  // Database
  getIt.registerLazySingleton<AppDatabase>(() => AppDatabase());

  // Services
  getIt.registerLazySingleton<NotificationService>(() => NotificationService());
  getIt.registerLazySingleton<BackupService>(() => BackupService(getIt<AppDatabase>()));

  // Repositories
  getIt.registerLazySingleton<HabitsRepository>(() => HabitsRepository(getIt<AppDatabase>()));
  getIt.registerLazySingleton<TasksRepository>(() => TasksRepository(getIt<AppDatabase>()));
  getIt.registerLazySingleton<NotesRepository>(() => NotesRepository(getIt<AppDatabase>()));

  getIt.registerFactory<HabitsCubit>(() => HabitsCubit(getIt<HabitsRepository>()));
  getIt.registerFactory<TasksCubit>(() => TasksCubit(getIt<TasksRepository>()));
  getIt.registerFactory<NotesCubit>(
    () => NotesCubit(getIt<NotesRepository>(), getIt<NotificationService>()),
  );
  getIt.registerFactory<ReviewsCubit>(() => ReviewsCubit(getIt<ReviewsRepository>()));
  getIt.registerFactory<HeatmapCubit>(
    () => HeatmapCubit(
      habitsRepository: getIt<HabitsRepository>(),
      tasksRepository: getIt<TasksRepository>(),
    ),
  );
}

import 'package:daily/cubits/heatmap/heatmap_state.dart';
import 'package:daily/utils/app_theme.dart';

class CustomContributionHeatmap extends StatefulWidget {
  final Map<DateTime, HeatmapDayData> dayDataMap;
  final DateTime minDate;
  final DateTime maxDate;
  final DateTime today;
  final void Function(DateTime date)? onCellTap;

  const CustomContributionHeatmap({
    super.key,
    required this.dayDataMap,
    required this.minDate,
    required this.maxDate,
    required this.today,
    this.onCellTap,
  });

  @override
  State<CustomContributionHeatmap> createState() => _CustomContributionHeatmapState();
}

class _CustomContributionHeatmapState extends State<CustomContributionHeatmap> {
  late final ScrollController _scrollController;

  @override
  void initState() {
    super.initState();
    _scrollController = ScrollController();
    WidgetsBinding.instance.addPostFrameCallback((_) => _scrollToToday());
  }

  @override
  void didUpdateWidget(covariant CustomContributionHeatmap oldWidget) {
    super.didUpdateWidget(oldWidget);
    WidgetsBinding.instance.addPostFrameCallback((_) => _scrollToToday());
  }

  @override
  void dispose() {
    _scrollController.dispose();
    super.dispose();
  }

  void _scrollToToday() {
    if (!_scrollController.hasClients) return;
    final startDate = AppDateUtils.normalizeDate(widget.minDate);
    final alignedStart = startDate.subtract(Duration(days: startDate.weekday - 1));
    final todayNorm = AppDateUtils.normalizeDate(widget.today);
    final daysDiff = todayNorm.difference(alignedStart).inDays;
    final weekIndex = daysDiff ~/ 7;
    const colWidth = 20.0; // cellWidth (16) + spacing (4)
    final targetOffset = (weekIndex * colWidth) - 150;
    _scrollController.jumpTo(targetOffset.clamp(0.0, _scrollController.position.maxScrollExtent));
  }

  String _getMonthName(int month) {
    const months = ['Янв', 'Фев', 'Мар', 'Апр', 'Май', 'Июн', 'Июл', 'Авг', 'Сен', 'Окт', 'Ноя', 'Дек'];
    return months[(month - 1).clamp(0, 11)];
  }

  @override
  Widget build(BuildContext context) {
    const double cellSize = 16.0;
    const double cellSpacing = 4.0;
    const double colWidth = cellSize + cellSpacing;

    final startDate = AppDateUtils.normalizeDate(widget.minDate);
    final alignedStart = startDate.subtract(Duration(days: startDate.weekday - 1));
    final endDate = AppDateUtils.normalizeDate(widget.maxDate);

    final List<List<DateTime?>> weeks = [];
    DateTime current = alignedStart;

    while (!current.isAfter(endDate) || weeks.length * 7 < (endDate.difference(alignedStart).inDays + 7)) {
      final List<DateTime?> week = [];
      for (int i = 0; i < 7; i++) {
        week.add(current);
        current = current.add(const Duration(days: 1));
      }
      weeks.add(week);
      if (current.isAfter(endDate) && weeks.length * 7 >= endDate.difference(alignedStart).inDays + 7) {
        break;
      }
    }

    int? lastMonth;
    int? lastYear;

    return SingleChildScrollView(
      controller: _scrollController,
      scrollDirection: Axis.horizontal,
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: weeks.map((week) {
          final firstDay = week.firstWhere((d) => d != null, orElse: () => null);
          String headerLabel = '';
          bool isYearLabel = false;
          if (firstDay != null) {
            if (firstDay.month == 1) {
              if (lastYear == null || firstDay.year != lastYear) {
                headerLabel = '${firstDay.year}';
                lastYear = firstDay.year;
                lastMonth = 1;
                isYearLabel = true;
              }
            } else {
              if (lastMonth == null || firstDay.month != lastMonth) {
                lastMonth = firstDay.month;
                headerLabel = _getMonthName(firstDay.month);
                if (lastYear == null || firstDay.year != lastYear) {
                  headerLabel = '${firstDay.year}';
                  lastYear = firstDay.year;
                  lastMonth = 1;
                  isYearLabel = true;
                }
              }
            }
          }

          return Container(
            width: colWidth,
            margin: const EdgeInsets.only(right: cellSpacing),
            child: Column(
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                // Single-line header label (Year for January, Month for Feb-Dec)
                SizedBox(
                  height: 16,
                  child: Text(
                    headerLabel,
                    style: TextStyle(
                      fontSize: isYearLabel ? 10 : 9,
                      fontWeight: isYearLabel ? FontWeight.bold : FontWeight.normal,
                      color: isYearLabel ? AppColors.textPrimary : AppColors.textSecondary,
                      height: 1.0,
                    ),
                    overflow: TextOverflow.visible,
                    softWrap: false,
                  ),
                ),
                const SizedBox(height: 2),
                // 7 Day square cells
                ...week.map((date) {
                  if (date == null) {
                    return Container(
                      width: cellSize,
                      height: cellSize,
                      margin: const EdgeInsets.only(bottom: cellSpacing),
                    );
                  }

                  final normDate = AppDateUtils.normalizeDate(date);
                  final dayData = widget.dayDataMap[normDate];
                  final target = dayData?.targetCount ?? 0;
                  final actual = dayData?.actualCount ?? 0;
                  final isToday = AppDateUtils.isSameDay(normDate, widget.today);

                  // Completion fraction (bottom to top fill)
                  final double fraction = target > 0
                      ? (actual / target).clamp(0.0, 1.0)

class _WaveClipper extends CustomClipper<Path> {
  const _WaveClipper({
    required this.center,
    required this.progress,
  });

  final Offset center;
  final double progress;

  @override
  Path getClip(Size size) {
    final dx = math.max(center.dx, size.width - center.dx);
    final dy = math.max(center.dy, size.height - center.dy);

    // Distance from the tap point to the farthest corner.
    final maxRadius = math.sqrt(dx * dx + dy * dy);
    final radius = maxRadius * progress;

    return Path()
      ..addOval(
        Rect.fromCircle(
          center: center,
          radius: radius,
        ),
      );
  }

  @override
  bool shouldReclip(covariant _WaveClipper oldClipper) {
    return oldClipper.center != center ||
        oldClipper.progress != progress;
  }
}

class HabitsCubit extends Cubit<HabitsState> {

  StreamSubscription? _habitsSub;
  StreamSubscription? _completionsSub;
  DateTime? _currentDate;

  HabitsCubit(this.repository) : super(HabitsInitial());

  /// Initializes streams for habits and completions for a specific date (defaults to today).
  void loadHabits([DateTime? date]) {
    _currentDate = date != null ? AppDateUtils.normalizeDate(date) : AppDateUtils.today;
    emit(HabitsLoading());
    _subscribe(_currentDate!);
  }

  void _subscribe(DateTime targetDate) {
    _habitsSub?.cancel();
    _completionsSub?.cancel();

    _habitsSub = repository.watchAllHabits().listen((allHabits) {
      final activeHabits = allHabits.where((h) => h.canceledAt == null).toList();

      _completionsSub?.cancel();
      _completionsSub = repository.watchCompletionsForDate(targetDate).listen((completions) {
        final completionsMap = <int, bool>{};
        for (final c in completions) {
          completionsMap[c.habitId] = c.isCompleted;
        }

        emit(HabitsLoaded(
          activeHabits: activeHabits,
          allHabits: allHabits,
          todayCompletions: completionsMap,
        ));
      }, onError: (e) {
        emit(HabitsError('Ошибка загрузки статуса привычек: $e'));
      });
    }, onError: (e) {
      emit(HabitsError('Ошибка загрузки привычек: $e'));
    });
  }

  Future<void> toggleCompletion(int habitId, DateTime date, bool isCompleted) async {
    try {
      final normDate = AppDateUtils.normalizeDate(date);
      await repository.toggleHabitCompletion(habitId, normDate, isCompleted);

      emit(HabitsError('Не удалось обновить статус привычки: $e'));
    }

      await repository.createHabit(title, description);
    } catch (e) {
      emit(HabitsError('Не удалось создать привычку: $e'));
    }
  }

  Future<void> updateHabit(int id, String title, String? description) async {

      await repository.updateHabit(id, title, description);
    } catch (e) {
      emit(HabitsError('Не удалось обновить привычку: $e'));

      emit(HabitsError('Не удалось восстановить привычку: $e'));
    }
  }

    _habitsSub?.cancel();
    _completionsSub?.cancel();
    return super.close();
  }

import 'dart:async';

import 'package:daily/cubits/heatmap/heatmap_state.dart';
import 'package:daily/data/repositories/habits_repository.dart';
import 'package:daily/data/repositories/tasks_repository.dart';
import 'package:daily/utils/date_utils.dart';
import 'package:flutter_bloc/flutter_bloc.dart';

class HeatmapCubit extends Cubit<HeatmapState> {
  final HabitsRepository habitsRepository;
  final TasksRepository tasksRepository;

  HeatmapMode _currentMode = HeatmapMode.habitsOnly;

  HeatmapCubit({
    required this.habitsRepository,
    required this.tasksRepository,
  }) : super(HeatmapInitial());

  /// Sets active heatmap display mode.
  void setMode(HeatmapMode mode) {
    _currentMode = mode;
    loadHeatmapData();
  }

  /// Calculates target vs actual day data for habits and tasks from minDate to future maxDate.
  Future<void> loadHeatmapData() async {
    if (state is! HeatmapLoaded) {
      emit(HeatmapLoading());
    }
    try {
      final today = AppDateUtils.today;
      final minDate = DateTime(today.year, today.month - 5, 1);

      final allHabits = await habitsRepository.watchAllHabits().first;
      final allCompletions = await habitsRepository.getAllCompletions();
      final allTasks = await tasksRepository.getAllTasks();

      // Determine max date: at least today + 14 days, or max task dueDate / future date up to 2 years
      DateTime maxDate = today.add(const Duration(days: 14));
      for (final t in allTasks) {
        if (t.dueDate != null) {
          final normDue = AppDateUtils.normalizeDate(t.dueDate!);
          if (normDue.isAfter(maxDate)) {
            maxDate = normDue;
          }
        }
      }
      final twoYearsLater = DateTime(today.year + 2, today.month, today.day);
      if (maxDate.isAfter(twoYearsLater)) {
        maxDate = twoYearsLater;
      }

      // Build completions lookup: habitId_dateIso -> isCompleted
      final completionsMap = <String, bool>{};
      for (final c in allCompletions) {
        if (c.isCompleted) {
          final key = '${c.habitId}_${AppDateUtils.normalizeDate(c.date).toIso8601String()}';
          completionsMap[key] = true;
        }
      }

      // Build tasks lookup: date -> target/actual counts
      final taskTargetsMap = <DateTime, int>{};
      final taskActualsMap = <DateTime, int>{};
      for (final t in allTasks) {
        if (t.dueDate != null) {
          final normDue = AppDateUtils.normalizeDate(t.dueDate!);
          taskTargetsMap[normDue] = (taskTargetsMap[normDue] ?? 0) + 1;
          if (t.isCompleted) {
            taskActualsMap[normDue] = (taskActualsMap[normDue] ?? 0) + 1;
          }
        }
      }

      final dayDataMap = <DateTime, HeatmapDayData>{};
      DateTime current = AppDateUtils.normalizeDate(minDate);
      final normMax = AppDateUtils.normalizeDate(maxDate);

      while (!current.isAfter(normMax)) {
        final activeHabitsForDay = allHabits.where((h) {
          final isCreated = AppDateUtils.normalizeDate(h.createdAt).isBefore(current) ||
              AppDateUtils.isSameDay(h.createdAt, current);
          final isNotCanceled = h.canceledAt == null ||
              AppDateUtils.normalizeDate(h.canceledAt!).isAfter(current);
          return isCreated && isNotCanceled;
        }).toList();

        int target = activeHabitsForDay.length;
        int actual = 0;

        for (final h in activeHabitsForDay) {
          final key = '${h.id}_${current.toIso8601String()}';
          if (completionsMap[key] == true) {
            actual++;
          }
        }

        if (_currentMode == HeatmapMode.allTasks) {
          target += taskTargetsMap[current] ?? 0;
          actual += taskActualsMap[current] ?? 0;
        }

        dayDataMap[current] = HeatmapDayData(
          date: current,
          targetCount: target,
          actualCount: actual,
        );

        current = current.add(const Duration(days: 1));
      }

      emit(HeatmapLoaded(
        mode: _currentMode,
        dayDataMap: dayDataMap,
        minDate: minDate,
        maxDate: maxDate,
        today: today,
      ));
    } catch (e) {
      emit(HeatmapError('Ошибка расчета heatmap: $e'));
    }
  }
}

  NotesCubit(this.repository, this.notificationService) : super(NotesInitial());

  /// Loads notes with optional filters and sorting.
  void loadNotes({
    DateTime? createdAfter,
    DateTime? createdBefore,
    bool? hasNeedCheckDate,
    DateTime? checkAfter,
    DateTime? checkBefore,
    bool? sortByReminder,
  }) {
    _createdAfter = createdAfter ?? _createdAfter;
    _createdBefore = createdBefore ?? _createdBefore;
    _hasNeedCheckDate = hasNeedCheckDate ?? _hasNeedCheckDate;
    _checkAfter = checkAfter ?? _checkAfter;
    _checkBefore = checkBefore ?? _checkBefore;
    if (sortByReminder != null) {
      _sortByReminder = sortByReminder;
    }

    emit(NotesLoading());
    _subscribe();
  }

  void toggleSort() {
    _sortByReminder = !_sortByReminder;
    loadNotes();
  }

  void _subscribe() {
    _notesSub?.cancel();
    _recentSub?.cancel();

    _recentSub = repository.watchRecentNotes(3).listen((recent) {
      _notesSub?.cancel();
      _notesSub = repository.watchNotes(
        createdAfter: _createdAfter,
        createdBefore: _createdBefore,
        hasNeedCheckDate: _hasNeedCheckDate,
        checkAfter: _checkAfter,
        checkBefore: _checkBefore,
        sortByReminder: _sortByReminder,
      ).listen((notes) {
        emit(NotesLoaded(
          notes: notes,
          recentNotes: recent,
          createdAfter: _createdAfter,
          createdBefore: _createdBefore,
          hasNeedCheckDate: _hasNeedCheckDate,
          checkAfter: _checkAfter,
          checkBefore: _checkBefore,
          sortByReminder: _sortByReminder,
        ));
      }, onError: (e) {
        emit(NotesError('Ошибка загрузки заметок: $e'));
      });
    }, onError: (e) {
      emit(NotesError('Ошибка загрузки последних заметок: $e'));
    });
  }

  void clearFilters() {
    _createdAfter = null;
    _createdBefore = null;
    _hasNeedCheckDate = null;
    _checkAfter = null;
    _checkBefore = null;
    loadNotes();
  }

  Future<void> createNote(String title, String description, DateTime? needCheckDate) async {
    try {
      final noteId = await repository.createNote(title, description, needCheckDate);
      if (needCheckDate != null) {
        await notificationService.scheduleNoteNotification(
          noteId: noteId,
          title: 'Напоминание: $title',
          body: description,
          scheduledDate: needCheckDate,
        );
      }
    } catch (e) {
      emit(NotesError('Не удалось создать заметку: $e'));
    }
  }

  Future<void> updateNote(int id, String title, String description, DateTime? needCheckDate) async {
    try {
      await repository.updateNote(id, title, description, needCheckDate);
      await notificationService.cancelNoteNotification(id);
      if (needCheckDate != null) {
        await notificationService.scheduleNoteNotification(
          noteId: id,
          title: 'Напоминание: $title',
          body: description,
          scheduledDate: needCheckDate,
        );
      }
    } catch (e) {
      emit(NotesError('Не удалось обновить заметку: $e'));
    }
  }

  Future<void> deleteNote(int id) async {
    try {
      await repository.deleteNote(id);
      await notificationService.cancelNoteNotification(id);
    } catch (e) {
      emit(NotesError('Не удалось удалить заметку: $e'));
    }
  }

  @override
  Future<void> close() {
    _notesSub?.cancel();
    _recentSub?.cancel();
    return super.close();
  }

  StreamSubscription? _reviewSub;

  ReviewsCubit(this.repository) : super(ReviewsInitial());

  /// Loads review for specified date (defaults to today).
  void loadReviewForDate([DateTime? date]) {
    final targetDate = AppDateUtils.normalizeDate(date ?? AppDateUtils.today);
    final isEditable = AppDateUtils.isSameDay(targetDate, AppDateUtils.today);

    emit(ReviewsLoading());

    _reviewSub?.cancel();
    _reviewSub = repository.watchReviewForDate(targetDate).listen((review) {
      emit(ReviewsLoaded(
        review: review,
        date: targetDate,
        isEditable: isEditable,
      ));
    }, onError: (e) {
      emit(ReviewsError('Ошибка загрузки отзыва: $e'));
    });
  }

  /// Saves or updates review. Enforces `date == today` requirement.
  Future<void> saveReview(DateTime date, String description, int rating) async {
    final normalizedDate = AppDateUtils.normalizeDate(date);
    if (!AppDateUtils.isSameDay(normalizedDate, AppDateUtils.today)) {
      emit(const ReviewsError('Отзыв можно редактировать только в текущий день!'));
      return;
    }

    try {
      await repository.saveReview(normalizedDate, description, rating);
    } catch (e) {
      emit(ReviewsError('Не удалось сохранить отзыв: $e'));
    }
  }

  @override
  Future<void> close() {
    _reviewSub?.cancel();
    return super.close();
  }
}

import 'package:daily/data/database/app_database.dart';
import 'package:daily/utils/date_utils.dart';
import 'package:drift/drift.dart';

class HabitsRepository {
  final AppDatabase db;

  HabitsRepository(this.db);

  /// Streams active habits (canceledAt is null).
  Stream<List<Habit>> watchActiveHabits() {
    return (db.select(db.habits)
          ..where((tbl) => tbl.canceledAt.isNull())
          ..orderBy([(tbl) => OrderingTerm.desc(tbl.createdAt)]))
        .watch();
  }

  /// Streams all habits (both active and soft-deleted).
  Stream<List<Habit>> watchAllHabits() {
    return (db.select(db.habits)
          ..orderBy([(tbl) => OrderingTerm.desc(tbl.createdAt)]))
        .watch();
  }

  /// Creates a new habit template.
  Future<int> createHabit(String title, String? description) {
    return db.into(db.habits).insert(
          HabitsCompanion.insert(
            title: title,
            description: Value(description),
            createdAt: DateTime.now(),
          ),
        );
  }

  /// Updates an existing habit template.
  Future<bool> updateHabit(int id, String title, String? description) {
    return (db.update(db.habits)..where((tbl) => tbl.id.equals(id))).write(
      HabitsCompanion(
        title: Value(title),
        description: Value(description),
      ),
    ).then((rows) => rows > 0);
  }

  /// Soft deletes a habit (sets canceledAt to current time).
  Future<bool> cancelHabit(int id) {
    return (db.update(db.habits)..where((tbl) => tbl.id.equals(id))).write(
      HabitsCompanion(
        canceledAt: Value(DateTime.now()),
      ),
    ).then((rows) => rows > 0);
  }

  /// Restores a soft-deleted habit (clears canceledAt).
  Future<bool> restoreHabit(int id) {
    return (db.update(db.habits)..where((tbl) => tbl.id.equals(id))).write(
      const HabitsCompanion(
        canceledAt: Value(null),
      ),
    ).then((rows) => rows > 0);
  }

  /// Streams completions for a given date.
  Stream<List<HabitCompletion>> watchCompletionsForDate(DateTime date) {
    final normalizedDate = AppDateUtils.normalizeDate(date);
    return (db.select(db.habitCompletions)
          ..where((tbl) => tbl.date.equals(normalizedDate)))
        .watch();
  }

  /// Returns completions for a given date.
  Future<List<HabitCompletion>> getCompletionsForDate(DateTime date) {
    final normalizedDate = AppDateUtils.normalizeDate(date);
    return (db.select(db.habitCompletions)
          ..where((tbl) => tbl.date.equals(normalizedDate)))
        .get();
  }

  /// Returns completions in a date range.
  Future<List<HabitCompletion>> getCompletionsInRange(DateTime start, DateTime end) {
    final normStart = AppDateUtils.normalizeDate(start);
    final normEnd = AppDateUtils.normalizeDate(end);
    return (db.select(db.habitCompletions)
          ..where((tbl) => tbl.date.isBetweenValues(normStart, normEnd)))
        .get();
  }

  /// Returns all completions across all history.
  Future<List<HabitCompletion>> getAllCompletions() {
    return db.select(db.habitCompletions).get();
  }

  /// Toggles habit completion status for a specific date (Upsert with proper existence check).
  Future<void> toggleHabitCompletion(int habitId, DateTime date, bool isCompleted) async {
    final normalizedDate = AppDateUtils.normalizeDate(date);

    final existing = await (db.select(db.habitCompletions)
          ..where((tbl) => tbl.habitId.equals(habitId) & tbl.date.equals(normalizedDate)))
        .getSingleOrNull();

    if (existing != null) {
      await (db.update(db.habitCompletions)
            ..where((tbl) => tbl.id.equals(existing.id)))
          .write(HabitCompletionsCompanion(
        isCompleted: Value(isCompleted),
      ));
    } else {

            HabitCompletionsCompanion.insert(
              habitId: habitId,
              date: normalizedDate,
              isCompleted: Value(isCompleted),
            ),
          );
    }
  }

class NotesRepository {
  final AppDatabase db;

  NotesRepository(this.db);

  /// Streams notes with optional filtering and sorting options.
  Stream<List<Note>> watchNotes({
    DateTime? createdAfter,
    DateTime? createdBefore,
    bool? hasNeedCheckDate,
    DateTime? checkAfter,
    DateTime? checkBefore,
    bool sortByReminder = false,
  }) {
    final query = db.select(db.notes);

    if (sortByReminder) {
      query.orderBy([
        (tbl) => OrderingTerm.desc(tbl.needCheckDate),
        (tbl) => OrderingTerm.desc(tbl.createdAt),
      ]);
    } else {
      query.orderBy([
        (tbl) => OrderingTerm.desc(tbl.createdAt),
      ]);
    }

    if (createdAfter != null) {
      final normAfter = AppDateUtils.normalizeDate(createdAfter);
      query.where((tbl) => tbl.createdAt.isBiggerOrEqualValue(normAfter));
    }
    if (createdBefore != null) {
      final normBefore = AppDateUtils.normalizeDate(createdBefore).add(const Duration(days: 1));
      query.where((tbl) => tbl.createdAt.isSmallerThanValue(normBefore));
    }
    if (hasNeedCheckDate != null) {
      if (hasNeedCheckDate) {
        query.where((tbl) => tbl.needCheckDate.isNotNull());
      } else {
        query.where((tbl) => tbl.needCheckDate.isNull());
      }
    }
    if (checkAfter != null) {
      final normCheckAfter = AppDateUtils.normalizeDate(checkAfter);
      query.where((tbl) => tbl.needCheckDate.isBiggerOrEqualValue(normCheckAfter));
    }
    if (checkBefore != null) {
      final normCheckBefore = AppDateUtils.normalizeDate(checkBefore);
      query.where((tbl) => tbl.needCheckDate.isSmallerOrEqualValue(normCheckBefore));
    }

    return query.watch();
  }

  /// Gets all notes.
  Future<List<Note>> getAllNotes() {
    return (db.select(db.notes)..orderBy([(tbl) => OrderingTerm.desc(tbl.createdAt)])).get();
  }

  /// Gets recent notes limit by [count].
  Future<List<Note>> getRecentNotes(int count) {
    return (db.select(db.notes)
          ..orderBy([(tbl) => OrderingTerm.desc(tbl.createdAt)])
          ..limit(count))
        .get();
  }

  /// Streams recent notes limit by [count].
  Stream<List<Note>> watchRecentNotes(int count) {
    return (db.select(db.notes)
          ..orderBy([(tbl) => OrderingTerm.desc(tbl.createdAt)])
          ..limit(count))
        .watch();
  }

  /// Creates a new note.
  Future<int> createNote(String title, String description, DateTime? needCheckDate) {
    return db.into(db.notes).insert(
          NotesCompanion.insert(
            title: title,
            description: description,
            createdAt: DateTime.now(),
            needCheckDate: Value(needCheckDate),
          ),
        );
  }

  /// Updates an existing note.
  Future<bool> updateNote(int id, String title, String description, DateTime? needCheckDate) {
    return (db.update(db.notes)..where((tbl) => tbl.id.equals(id))).write(
      NotesCompanion(
        title: Value(title),
        description: Value(description),
        needCheckDate: Value(needCheckDate),
      ),
    ).then((rows) => rows > 0);
  }

  /// Deletes a note by id.
  Future<bool> deleteNote(int id) {
    return (db.delete(db.notes)..where((tbl) => tbl.id.equals(id))).go().then((rows) => rows > 0);
  }
}

  }

  /// Saves or updates daily review for a specific date.
  /// Throws Exception if trying to edit/save for a past date (`date < today`).

    }

    await db.into(db.dailyReviews).insertOnConflictUpdate(
          DailyReviewsCompanion.insert(
            date: normalizedDate,
            description: description,
            rating: rating.clamp(1, 10),
          ),
        );
  }

    if (showLoading && mounted) {

      final review = await _reviewsRepo.getReviewForDate(widget.date);

            ? null
            : AppDateUtils.normalizeDate(habit.canceledAt!);

        final isNotCanceled =
            canceledDate == null || canceledDate.isAfter(targetDate);

                      TextField(
                        controller: titleController,
                        decoration: const InputDecoration(labelText: 'Заголовок'),
                      ),
                      const SizedBox(height: 12),
                      TextField(
                        controller: descController,
                        maxLines: 4,
                        decoration: const InputDecoration(labelText: 'Текст заметки'),
                      ),
                      const SizedBox(height: 16),
                      Row(
                        children: [
                          Expanded(
                            child: Text(
                              needCheckDate == null
                                  ? 'Дата напоминания: не задана'
                                  : 'Напоминание: ${AppDateUtils.formatDateDisplay(needCheckDate!)}',
                              style: const TextStyle(fontSize: 13),
                            ),
                          ),
                          IconButton(
                            icon: Icon(
                              needCheckDate == null ? Icons.event_available : Icons.event_busy,
                              color: AppColors.primary,
                            ),
                            onPressed: () async {
                              if (needCheckDate != null) {
                                setDialogState(() => needCheckDate = null);
                              } else {
                                final pickedDate = await showDatePicker(
                                  context: context,
                                  initialDate: DateTime.now().add(const Duration(days: 1)),
                                  firstDate: DateTime.now(),
                                  lastDate: DateTime.now().add(const Duration(days: 365)),
                                );
                                if (pickedDate != null && context.mounted) {
                                  final pickedTime = await showTimePicker(
                                    context: context,
                                    initialTime: const TimeOfDay(hour: 9, minute: 0),
                                  );
                                  if (pickedTime != null) {
                                    setDialogState(() {
                                      needCheckDate = DateTime(
                                        pickedDate.year,
                                        pickedDate.month,
                                        pickedDate.day,
                                        pickedTime.hour,
                                        pickedTime.minute,
                                      );
                                    });
                                  }
                                }
                              }
                            },
                          ),
                        ],
                      ),
                    ],
                  ),
                ),
                actions: [
                  TextButton(
                    onPressed: () => Navigator.pop(dialogContext),
                    child: const Text('Отмена'),
                  ),
                  ElevatedButton(
                    onPressed: () {
                      final title = titleController.text.trim();
                      final desc = descController.text.trim();

                      if (title.isNotEmpty) {
                        final cubit = context.read<NotesCubit>();
                        if (note == null) {
                          cubit.createNote(title, desc, needCheckDate);
                        } else {
                          cubit.updateNote(note.id, title, desc, needCheckDate);
                        }
                        Navigator.pop(dialogContext);
                      }
                    },
                    style: ElevatedButton.styleFrom(
                      backgroundColor: AppColors.primary,
                      foregroundColor: Colors.black,
                    ),
                    child: const Text('Сохранить'),
                  ),
                ],
              );
            },
          ),
        );
      },
    );
  }

  void _showFilterSheet(BuildContext context) {
    final notesCubit = context.read<NotesCubit>();
    showModalBottomSheet(
      context: context,
      builder: (sheetContext) {
        return BlocProvider.value(
          value: notesCubit,
          child: StatefulBuilder(
            builder: (context, setSheetState) {
              return Padding(
                padding: const EdgeInsets.all(16),
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Text('Фильтры заметок',
                        style: TextStyle(fontSize: 18, fontWeight: FontWeight.bold)),
                    const SizedBox(height: 16),
                    Row(
                      children: [
                        const Text('Напоминание:'),
                        const SizedBox(width: 12),
                        DropdownButton<bool?>(
                          value: _filterHasCheckDate,
                          items: const [
                            DropdownMenuItem(value: null, child: Text('Все')),
                            DropdownMenuItem(value: true, child: Text('Есть напоминание')),
                            DropdownMenuItem(value: false, child: Text('Без напоминания')),
                          ],
                          onChanged: (val) {
                            setSheetState(() => _filterHasCheckDate = val);
                          },
                        ),
                      ],
                    ),
                    const SizedBox(height: 12),
                    Row(
                      children: [
                        Expanded(
                          child: Text(_filterCreatedAfter == null
                              ? 'Дата создания с: Все'
                              : 'С: ${AppDateUtils.formatDateDisplay(_filterCreatedAfter!)}'),
                        ),
                        TextButton(
                          onPressed: () async {
                            final picked = await showDatePicker(
                              context: context,
                              initialDate: DateTime.now(),
                              firstDate: DateTime(2020),
                              lastDate: DateTime.now(),
                            );
                            if (picked != null) {
                              setSheetState(() => _filterCreatedAfter = picked);
                            }
                          },
                          child: const Text('Выбрать'),
                        ),
                      ],
                    ),
                    Row(
                      mainAxisAlignment: MainAxisAlignment.end,
                      children: [
                        TextButton(
                          onPressed: () {
                            setState(() {
                              _filterCreatedAfter = null;
                              _filterCreatedBefore = null;
                              _filterHasCheckDate = null;
                            });
                            context.read<NotesCubit>().clearFilters();
                            Navigator.pop(sheetContext);
                          },
                          child: const Text('Сбросить'),
                        ),
                        ElevatedButton(
                          onPressed: () {
                            setState(() {});
                            context.read<NotesCubit>().loadNotes(
                                  createdAfter: _filterCreatedAfter,
                                  createdBefore: _filterCreatedBefore,
                                  hasNeedCheckDate: _filterHasCheckDate,
                                );
                            Navigator.pop(sheetContext);
                          },
                          style: ElevatedButton.styleFrom(
                            backgroundColor: AppColors.primary,
                            foregroundColor: Colors.black,
                          ),
                          child: const Text('Применить'),
                        ),
                      ],
                    ),
                  ],
                ),
              );
            },
          ),
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Заметки'),
        actions: [
          BlocBuilder<NotesCubit, NotesState>(
            builder: (context, state) {
              final sortByReminder = state is NotesLoaded && state.sortByReminder;
              return IconButton(
                icon: Icon(
                  sortByReminder ? Icons.event : Icons.access_time,
                  color: AppColors.primary,
                ),
                tooltip: sortByReminder ? 'Сортировка: по дате напоминания' : 'Сортировка: по дате создания',
                onPressed: () {
                  context.read<NotesCubit>().toggleSort();
                },
              );
            },
          ),
          IconButton(
            icon: const Icon(Icons.filter_list),
            onPressed: () => _showFilterSheet(context),
          ),
        ],
      ),
      floatingActionButton: FloatingActionButton(
        onPressed: () => _showAddEditDialog(context),
        child: const Icon(Icons.add),
      ),
      body: BlocConsumer<NotesCubit, NotesState>(
        listener: (context, state) {
          if (state is NotesError) {
            ScaffoldMessenger.of(context).showSnackBar(
              SnackBar(content: Text(state.message)),
            );
          }
        },
        builder: (context, state) {
          if (state is NotesLoading) {
            return const Center(child: CircularProgressIndicator());
          }

          if (state is NotesLoaded) {
            if (state.notes.isEmpty) {
              return const Center(
                child: Text('Заметок пока нет.', style: TextStyle(color: AppColors.textSecondary)),
              );
            }

            return ListView.builder(
              padding: const EdgeInsets.all(16),
              itemCount: state.notes.length,
              itemBuilder: (context, index) {
                final note = state.notes[index];
                return Card(
                  margin: const EdgeInsets.only(bottom: 16),
                  child: Padding(
                    padding: const EdgeInsets.all(12),
                    child: Column(
                      crossAxisAlignment: CrossAxisAlignment.start,
                      children: [
                        Row(
                          mainAxisAlignment: MainAxisAlignment.spaceBetween,
                          children: [
                            Expanded(
                              child: Text(
                                note.title,
                                style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16),
                              ),
                            ),
                            PopupMenuButton<String>(
                              onSelected: (val) {
                                if (val == 'edit') {
                                  _showAddEditDialog(context, note: note);
                                } else if (val == 'delete') {
                                  context.read<NotesCubit>().deleteNote(note.id);
                                }
                              },
                              itemBuilder: (context) => [
                                const PopupMenuItem(value: 'edit', child: Text('Редактировать')),
                                const PopupMenuItem(
                                  value: 'delete',
                                  child: Text('Удалить', style: TextStyle(color: Colors.red)),
                                ),
                              ],
                            ),
                          ],
                        ),
                        const SizedBox(height: 6),
                        Text(
                          note.description,
                          style: const TextStyle(color: AppColors.textSecondary),
                        ),
                        const SizedBox(height: 12),
                        Row(
                          children: [
                            Text(
                              'Создано: ${AppDateUtils.formatDateDisplay(note.createdAt)}',
                              style: const TextStyle(fontSize: 11, color: Colors.grey),
                            ),
                            if (note.needCheckDate != null) ...[
                              const Spacer(),
                              const Icon(Icons.notifications_active, size: 14, color: AppColors.primary),
                              const SizedBox(width: 4),
                              Text(
                                AppDateUtils.formatDateDisplay(note.needCheckDate!),
                                style: const TextStyle(fontSize: 11, color: AppColors.primary),
                              ),
                            ],
                          ],
                        ),
                      ],
                    ),
                  ),
                );
              },
            );
          }

          return const SizedBox.shrink();
        },
      ),
    );
  }
